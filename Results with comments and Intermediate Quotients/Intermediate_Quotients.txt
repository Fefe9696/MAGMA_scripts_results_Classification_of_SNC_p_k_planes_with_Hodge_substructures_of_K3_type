// This script takes in input a group G\cong Z_p^k and a building data v of a G-cover of P2 and  it returns the invariants of any intermediate quotient. 

load"Computation_BuildingData_up_to_Iso.txt";
load"Strict_Transforms_Comp_Branch_locus.txt";

Intermediate_Quotients:=function(v,G) 
  G,psi:=AbelianGroup(G);
  v:=[<psi(v[j][1]),v[j][2]>: j in [1..#v]];  
  
  p:=Exponent(G);
  k:=#Generators(G);
  S:=Subgroups(G);
  Cases:=[];
  for t in [1..k] do 
   T:=AbelianGroup([p : f in [1..t]]);
   Data:=[*[* Sprintf("Subgroups H of G of order %o^%o", Exponent(G), k-t), T*], [**]*];
   //Append(~Data, [* Sprintf("Subgroups H of G of order %o", p^(k-t)), T*]); 
    for j in [1..#S] do 
    H := S[j]`subgroup;
    if Order(G)/Order(H) eq p^t then 
     Q, pi:=quo<G|H>;
     if #Generators(H) ne 0 then 
      Gens:=<(G!g) @@ psi : g in Generators(H)>;
     else 
      Gens:=<Id(G) @@ psi>; 
     end if; 
     is_iso, iso := IsIsomorphic(Q, T); 
     Append(~Data[2], [*Gens, Q, pi, iso*]); 
     end if; 
   end for;
   Append(~Cases, Data);
   end for; 

  Int_Quo:=[**];
  for Data in Cases do 
     List:=[*[*Data[1]*], [*{Data[2][1][1]}*]*];
     T:=Data[1][2]; 
  
     for x in Data[2] do 
      for n in [1..#List[2]] do 
       Test:=true; 
      for y in List[2][n] do 
         for l in Data[2] do if l[1] eq y then Y:=l; break l; end if; end for;
         if AreEquivalent([<x[4](x[3](v[j][1])),v[j][2]>: j in [1..#v]],[<Y[4](Y[3](v[j][1])),v[j][2]>: j in [1..#v]],T) then 
             Include(~List[2][n], x[1]); Test:=false; 
             break n; 
         else break y; 
         end if; 
      end for;  
     end for; 
    if Test then 
     Append(~List[2], {x[1]}); 
    end if;
    end for; 
    Append(~Int_Quo, List); 
   end for; 

   return Int_Quo; 
end function;
