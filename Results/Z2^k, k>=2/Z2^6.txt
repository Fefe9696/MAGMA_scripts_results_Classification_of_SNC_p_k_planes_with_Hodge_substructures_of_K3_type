load"Main_Script_Classification_Algorithm.txt";

k:=6; 

G:=AbelianGroup([2: j in [1..k]]);



for sum_dg in [Maximum(6,k+2)..Floor(12-2^(4-k))] do 
 for card in [k..sum_dg] do 
   if k ne 6 then 
    Res:=Building_Data(card,sum_dg,G); 
    if not IsEmpty(Res) then Res; 
    end if;
   else 
     if <card,sum_dg> in {<8,8>,<9,9>,<10,10>,<11,11>} then 
       Res:=Building_Data(card,sum_dg,G); 
       if not IsEmpty(Res) then Res; 
       end if;
     end if;  
   end if; 
 end for;  
end for;
