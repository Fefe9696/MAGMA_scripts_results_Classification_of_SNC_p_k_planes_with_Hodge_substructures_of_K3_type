load"Main_Script_Classification_Algorithm.txt";

k:=4; 


G:=AbelianGroup([3: j in [1..k]]);

for sum_dg in [Maximum(5,k+2)..Floor(5/2*(3-1/(3^(k-1))) )] do 
 for card in [k..sum_dg] do  
    Res:=Building_Data(card,sum_dg,G); 
    if not IsEmpty(Res) then Res; 
    end if;
 end for;  
end for;
