// From Proposition 2.2 we have that card<=Min(p-1,sum_dg), and sum_dg=4. 

G:=AbelianGroup([17]);

p:=Order(G);
sum_dg:=4;

for card in [1..Minimum(p-1,sum_dg)] do 
   Res:=Building_Data(card,4,G); 
   if not IsEmpty(Res) then Res; end if; 
end for;

    ]
}
