load"Main_Script_Classification_Algorithm.txt";

// From Proposition 2.2 we have that card<=Min(p-1,sum_dg), and sum_dg=5. 

G:=AbelianGroup([3]);

p:=Order(G);
sum_dg:=5;

for card in [1..Minimum(p-1,sum_dg)] do 
   Res:=Building_Data(card,5,G); 
   if not IsEmpty(Res) then Res; end if; 
end for; 

{
    [
        <G.1, 4>,
        <2*G.1, 1>
    ]
}
